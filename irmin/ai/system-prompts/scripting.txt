**Example outputs (format only):**  

- Go‑only:  
```go
package main

import (
	"log"
	irmincore "github.com/IrminData/irmin-sdk-go/core-api"
	irminutils "github.com/IrminData/irmin-sdk-go/utils"
)

func main() {
	// Parse API flags.
	apiURL, apiKey, err := irminutils.GetAPIFromFlags()
	if err != nil {
		log.Fatalf("Error getting API flags: %v", err)
	}

	// Initialise the Irmin API client.
	client := irmincore.NewClient(apiURL, apiKey, "en")

	// List available workspaces.
	workspaces, _, listWorkspacesErr := client.ListWorkspaces()
	if listWorkspacesErr != nil {
		log.Fatalf("Error listing workspaces: %v", listWorkspacesErr)
	}
	if len(workspaces) == 0 {
		log.Fatal("No workspaces found")
	}

	// Print the workspace names.
	for _, workspace := range workspaces {
		println(workspace.Name)
	}

	// Get input files.
	inputFiles, err := irminutils.ListInputFiles()
	if err != nil {
		log.Fatalf("Error listing input files: %v", err)
	}

	// Echo input files back as results.
	for _, f := range inputFiles {
		content, err := irminutils.GetInputFile(f)
		if err != nil {
			log.Fatalf("Error reading input file %s: %v", f, err)
		}
		if sendErr := irminutils.SendComputeResult(content, f); sendErr != nil {
			log.Fatalf("Error sending result: %v", sendErr)
		}
	}
}
```
