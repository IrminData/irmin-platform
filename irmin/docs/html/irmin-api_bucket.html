Package irmin-api/bucket

package bucket // import "irmin-api/bucket"


TYPES

type Client struct {
	fiberS3.Storage
	Bucket   string
	Endpoint string
	Region   string
	Env      *utils.CoreAPIEnv
	DB       *db.Database
}
    Client is a client for interacting with S3 for storing and retrieving files.

func CreateClient(env *utils.CoreAPIEnv, bucketName string, database *db.Database) (*Client, error)
    CreateClient creates a new S3 client.

func (bucket *Client) DeletePath(ctx context.Context, keyPrefix string, tx ...*gorm.DB) error

func (bucket *Client) DownloadFolder(ctx context.Context, folderPrefix, localDir string) error
    DownloadFolder downloads all objects under the given folder prefix from
    the S3 bucket and saves them to the specified local directory. It creates
    any necessary local directories. - ctx: the context for the request -
    folderPrefix: the prefix for the folder in the bucket - localDir: the local
    directory where files will be saved Returns an error if any step fails.

func (bucket *Client) DuplicatePath(
	ctx context.Context,
	sourceKey, destKey string,
	removeOriginal bool,
	tx ...*gorm.DB,
) error

func (bucket *Client) ListObjects(ctx context.Context, keyPrefix string) ([]types.Object, error)

func (bucket *Client) PresignedGetURL(ctx context.Context, key string, expiry time.Duration) (string, error)
    PresignedGetURL generates a presigned GET URL for the given S3 key that
    expires after the given duration.

func (bucket *Client) ReadPath(ctx context.Context, key string) (*string, error)

func (bucket *Client) UploadReader(ctx context.Context, key string, body io.Reader) error
    UploadReader uploads the contents of an io.Reader to the given S3 key.

func (bucket *Client) WritePath(ctx context.Context, key string, content string, tx ...*gorm.DB) error
