package client // import "irmin-connectors/connectors/pinecone/client"


CONSTANTS

const (
	// UpsertBatchSize is the maximum number of vectors to upsert in a single batch.
	UpsertBatchSize = 100
	// FetchBatchSize is the maximum number of vectors to fetch in a single batch.
	// Pinecone ListVectors API limit is 100.
	FetchBatchSize = 100
	// DefaultTopK is the default number of results for search queries.
	DefaultTopK = 10
	// MaxTopK is the maximum allowed value for topK parameter (Pinecone API limit).
	MaxTopK = 10000
)

FUNCTIONS

func ValidateAPIKey(apiKey string) error
    ValidateAPIKey tests the API key by listing available indexes. This
    validates that the API key is valid without requiring a specific index host.

func ValidateConnection(apiKey, host, namespace string) error
    ValidateConnection tests the connection to a specific Pinecone index.


TYPES

type EmbeddingRecord struct {
	ID         string            `json:"id"`
	SourceFile string            `json:"source_file"`
	ChunkIndex int               `json:"chunk_index"`
	Content    string            `json:"content"`
	Embedding  []float32         `json:"embedding"`
	Metadata   map[string]string `json:"metadata"`
	CreatedAt  time.Time         `json:"created_at"`
}
    EmbeddingRecord represents the Irmin embedding format.

type PineconeClient struct {
	// Has unexported fields.
}
    PineconeClient wraps the Pinecone SDK client for connector operations.

func InitPineconeClient(
	_ any,
	logger *slog.Logger,
	operation *db.Operation,
	opts ...PineconeOption,
) (*PineconeClient, *string, error)
    InitPineconeClient initializes a Pinecone client from operation
    configuration.

func (c *PineconeClient) Close() error
    Close releases resources held by the client.

func (c *PineconeClient) Delete(ctx context.Context, ids []string) error
    Delete removes vectors from the Pinecone index by their IDs.

func (c *PineconeClient) FetchAll(ctx context.Context) ([]EmbeddingRecord, error)
    FetchAll fetches all vectors from the index/namespace.

func (c *PineconeClient) Search(ctx context.Context, queryVector []float32, topK int) (*SearchResponse, error)
    Search performs a similarity search using a query vector.

func (c *PineconeClient) Upsert(ctx context.Context, records []EmbeddingRecord) error
    Upsert inserts or updates vectors in the Pinecone index.

func (c *PineconeClient) UpsertSingle(ctx context.Context, record EmbeddingRecord) error
    UpsertSingle inserts or updates a single vector in the Pinecone index. This
    is a convenience method for patch operations that operate on single records.

type PineconeOption func(*PineconeClient)
    PineconeOption configures the PineconeClient at construction
    time. Mirrors the functional-options shape Stripe uses
    (connectors/stripe/client/client.go) so connector authors can copy either as
    a reference.

func WithProgressHandler(h common.ProgressHandler) PineconeOption
    WithProgressHandler installs an observability hook for the pagination +
    batching loops. Pass nil to disable (same as the default).

type SearchResponse struct {
	Query   string         `json:"query"`
	Matches []SearchResult `json:"matches"`
}
    SearchResponse represents the response from a search query.

type SearchResult struct {
	ID       string            `json:"id"`
	Score    float32           `json:"score"`
	Metadata map[string]string `json:"metadata,omitempty"`
}
    SearchResult represents a single search match from Pinecone.
